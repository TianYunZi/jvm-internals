package org.example.jvminternals.ch08;

import java.io.IOException;
import java.lang.classfile.ClassFile;
import java.lang.constant.ClassDesc;
import java.lang.constant.ConstantDescs;
import java.lang.constant.MethodTypeDesc;
import java.lang.invoke.MethodHandles;
import java.nio.file.Files;
import java.nio.file.Path;

/** 生成一份合法类文件，再用当前同包 Lookup 定义为非 STRONG 隐藏类。 */
public final class HiddenJourney {
    private static final System.Logger LOG = System.getLogger(HiddenJourney.class.getName());
    private HiddenJourney() { }

    public static void main(String[] args) throws IOException, ReflectiveOperationException {
        if (args.length != 1) {
            throw new IllegalArgumentException("用法：HiddenJourney <生成类文件输出路径>");
        }
        LOG.log(System.Logger.Level.INFO, "开始生成并定义一个隐藏类");
        byte[] bytes = ClassFile.of().build(ClassDesc.of("org.example.jvminternals.ch08.HiddenValue"),
                builder -> builder.withFlags(ClassFile.ACC_PUBLIC | ClassFile.ACC_FINAL)
                        .withMethodBody("value", MethodTypeDesc.of(ConstantDescs.CD_int),
                                ClassFile.ACC_PUBLIC | ClassFile.ACC_STATIC,
                                code -> code.bipush(42).ireturn()));
        Path output = Path.of(args[0]);
        Files.createDirectories(output.toAbsolutePath().getParent());
        Files.write(output, bytes);
        // 保存的是定义前字节；运行时隐藏名称由虚拟机产生，不写回原文件。
        Class<?> hidden = MethodHandles.lookup().defineHiddenClass(bytes, true,
                MethodHandles.Lookup.ClassOption.NESTMATE).lookupClass();
        System.out.println("name=" + hidden.getName() + " hidden=" + hidden.isHidden());
        System.out.println("loader=" + hidden.getClassLoader().getName()
                + " nest.host=" + hidden.getNestHost().getName());
        System.out.println("nominal.descriptor.present=" + hidden.describeConstable().isPresent());
        System.out.println("value=" + hidden.getDeclaredMethod("value").invoke(null));
        LOG.log(System.Logger.Level.INFO, "隐藏类观察结束");
    }
}

TypeJourney.java

package org.example.jvminternals.ch08;

import java.io.Serializable;
import java.lang.reflect.Field;
import java.lang.reflect.Method;
import java.util.Arrays;
import java.util.Comparator;

/** 正常观察类型身份、反射描述和数组关系，不猜测私有地址或对象偏移。 */
public final class TypeJourney {
    private static final System.Logger LOG = System.getLogger(TypeJourney.class.getName());
    private TypeJourney() { }

    interface Named { String name(); }
    static class Base { int id = 8; }
    static final class Item extends Base implements Named {
        private final String label = "item";
        @Override public String name() { return label; }
    }

    public static void main(String[] args) {
        LOG.log(System.Logger.Level.INFO, "开始类型和反射描述观察");
        Item item = new Item();
        Class<?> type = item.getClass();
        System.out.println("class=" + type.getName() + " mirror.class=" + type.getClass().getName());
        System.out.println("super=" + type.getSuperclass().getName() + " interfaces="
                + Arrays.toString(type.getInterfaces()));
        Field[] fields = type.getDeclaredFields();
        Arrays.sort(fields, Comparator.comparing(Field::getName));
        for (Field field : fields) {
            System.out.println("declared.field=" + field.getName() + " type=" + field.getType().getName());
        }
        Method[] methods = type.getDeclaredMethods();
        Arrays.sort(methods, Comparator.comparing(Method::getName));
        for (Method method : methods) {
            System.out.println("declared.method=" + method.getName() + " returns=" + method.getReturnType().getName());
        }
        System.out.println("item.isBase=" + Base.class.isInstance(item)
                + " item.isNamed=" + Named.class.isInstance(item));
        System.out.println("Object[].from.Item[]=" + Object[].class.isAssignableFrom(Item[].class));
        System.out.println("Named[].from.Item[]=" + Named[].class.isAssignableFrom(Item[].class));
        System.out.println("Object[].from.int[]=" + Object[].class.isAssignableFrom(int[].class));
        System.out.println("Object[].from.int[][]=" + Object[].class.isAssignableFrom(int[][].class));
        System.out.println("int[].isCloneable=" + Cloneable.class.isAssignableFrom(int[].class)
                + " int[].isSerializable=" + Serializable.class.isAssignableFrom(int[].class));
        System.out.println("Item[].component=" + Item[].class.getComponentType().getName()
                + " int[].component=" + int[].class.getComponentType().getName());
        LOG.log(System.Logger.Level.INFO, "类型和反射描述观察结束");
    }
}

UnloadingJourney.java

正式路径：src/main/java/org/example/jvminternals/ch08/UnloadingJourney.java

package org.example.jvminternals.ch08;

import java.io.IOException;
import java.lang.management.ManagementFactory;
import java.lang.management.MemoryPoolMXBean;
import java.lang.ref.ReferenceQueue;
import java.lang.ref.WeakReference;
import java.net.URL;
import java.net.URLClassLoader;
import java.nio.file.Path;

/** 先持有实例再释放；只请求两次回收，记录实际结果而不要求卸载必然发生。 */
public final class UnloadingJourney {
    private static final System.Logger LOG = System.getLogger(UnloadingJourney.class.getName());
    private static volatile Object retained;
    private UnloadingJourney() { }

    private static WeakReference<ClassLoader> createAndRetain(Path directory,
            ReferenceQueue<ClassLoader> queue) throws IOException, ReflectiveOperationException {
        URL url = directory.toUri().toURL();
        try (URLClassLoader loader = new URLClassLoader("disposable", new URL[] {url},
                ClassLoader.getPlatformClassLoader())) {
            Class<?> type = Class.forName("org.example.jvminternals.ch08.disposable.Disposable", true, loader);
            retained = type.getConstructor().newInstance();
            System.out.println("retained.class=" + retained.getClass().getName()
                    + " loader=" + loader.getName());
            return new WeakReference<>(loader, queue);
        }
        // 返回后本方法的 Class 与加载器局部变量离开栈帧，资源已关闭，实例仍被静态字段持有。
    }

    private static void printUsage(MemoryPoolMXBean pool, String phase) {
        var usage = pool.getUsage();
        System.out.println(phase + " metaspace.used=" + usage.getUsed()
                + " committed=" + usage.getCommitted());
    }

    public static void main(String[] args) throws IOException, ReflectiveOperationException, InterruptedException {
        if (args.length != 1) {
            throw new IllegalArgumentException("用法：UnloadingJourney <隔离类输出目录>");
        }
        LOG.log(System.Logger.Level.INFO, "开始有限加载器持有与释放观察");
        MemoryPoolMXBean pool = ManagementFactory.getMemoryPoolMXBeans().stream()
                .filter(bean -> "Metaspace".equals(bean.getName())).findFirst().orElseThrow();
        printUsage(pool, "before.load");
        ReferenceQueue<ClassLoader> queue = new ReferenceQueue<>();
        WeakReference<ClassLoader> reference = createAndRetain(Path.of(args[0]), queue);
        printUsage(pool, "after.load");
        System.gc();
        System.out.println("after.held.request loader.cleared=" + reference.refersTo(null));
        printUsage(pool, "after.held.request");
        retained = null;
        LOG.log(System.Logger.Level.INFO, "已释放教学静态持有者，仅再请求一次回收");
        System.gc();
        // 队列等待至多两秒；没有事件也是有效观察，不反复制造分配负载。
        System.out.println("after.release loader.cleared=" + reference.refersTo(null)
                + " queue.event=" + (queue.remove(2_000) != null));
        printUsage(pool, "after.release");
        LOG.log(System.Logger.Level.INFO, "加载器持有与释放观察结束");
    }
}
