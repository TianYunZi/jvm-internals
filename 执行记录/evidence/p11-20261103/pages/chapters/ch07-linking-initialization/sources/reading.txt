package org.example.jvminternals.ch07;

/** 已授权教学故障：同一类首次初始化失败和后续主动使用。 */
public final class InitializationFailure {
    private static final System.Logger LOG = System.getLogger(InitializationFailure.class.getName());
    private InitializationFailure() { }

    static final class Broken {
        static final int VALUE = fail();
        private static int fail() {
            throw new IllegalStateException("教学初始化失败");
        }
    }

    public static void main(String[] args) throws ClassNotFoundException {
        LOG.log(System.Logger.Level.INFO, "开始两次有界失败访问");
        Class<?> discovered = Class.forName(Broken.class.getName(), false,
                InitializationFailure.class.getClassLoader());
        System.out.println("discovered=" + discovered.getName());
        for (int attempt = 1; attempt <= 2; attempt++) {
            try {
                System.out.println("value=" + Broken.VALUE);
            } catch (ExceptionInInitializerError | NoClassDefFoundError failure) {
                System.out.println("attempt=" + attempt + " error=" + failure);
                Throwable cause = failure.getCause();
                // 原因链最多三层，保留错误传播证据而不无限展开日志。
                for (int depth = 1; cause != null && depth <= 3; depth++) {
                    System.out.println("cause[" + depth + "]=" + cause);
                    cause = cause.getCause();
                }
            }
        }
        LOG.log(System.Logger.Level.INFO, "两次失败访问结束");
    }
}

package org.example.jvminternals.ch07;

/** 常量、运行期赋值、相关接口、递归和实际接收者的正常教学观察。 */
public final class InitializationJourney {
    private static final System.Logger LOG = System.getLogger(InitializationJourney.class.getName());

    private InitializationJourney() { }

    static int mark(String phase, int value) {
        System.out.println("initialize=" + phase + " value=" + value);
        return value;
    }

    public static final class Values {
        public static final int CONSTANT = 41;
        public static final int RUNTIME = mark("Values.RUNTIME", 42);
        private Values() { }
    }

    interface HasDefault {
        int TOKEN = mark("HasDefault", 20);
        default int token() { return TOKEN; }
    }

    interface Plain {
        int TOKEN = mark("Plain", 30);
    }

    static class Parent {
        static final int VALUE = mark("Parent", 10);
    }

    static final class Child extends Parent implements HasDefault, Plain {
        static final int VALUE = mark("Child", 40);
    }

    static final class Recursive {
        static final int FIRST = readLater();
        static int later = 7;

        private static int readLater() {
            // 同一初始化线程递归调用，尚未执行的 later 赋值仍不可凭空出现。
            return later + 1;
        }
    }

    public static final class Lazy {
        public static int value = mark("Lazy", 50);
        private Lazy() { }
    }

    static class Base {
        int field = 1;
        int number() { return 1; }
    }

    static final class Derived extends Base {
        int field = 2;
        @Override int number() { return 2; }
    }

    public static void main(String[] args) throws ReflectiveOperationException {
        LOG.log(System.Logger.Level.INFO, "开始正常初始化观察");
        System.out.println("constant=" + Values.CONSTANT);
        System.out.println("before.runtime");
        System.out.println("runtime=" + Values.RUNTIME);
        System.out.println("child=" + Child.VALUE);
        System.out.println("plain=" + Plain.TOKEN);
        System.out.println("recursive.first=" + Recursive.FIRST + " later=" + Recursive.later);
        Class<?> discovered = Class.forName(Lazy.class.getName(), false,
                InitializationJourney.class.getClassLoader());
        System.out.println("discovered=" + discovered.getName());
        System.out.println("reflect.value=" + discovered.getField("value").get(null));
        Base receiver = new Derived();
        System.out.println("field=" + receiver.field + " virtual=" + receiver.number());
        LOG.log(System.Logger.Level.INFO, "正常初始化观察结束");
    }
}

InitializationRace.java

正式路径：src/main/java/org/example/jvminternals/ch07/InitializationRace.java

package org.example.jvminternals.ch07;

import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;

/** 两个线程请求同一类初始化；门闩只用于形成有界的人工观察窗口。 */
public final class InitializationRace {
    private static final System.Logger LOG = System.getLogger(InitializationRace.class.getName());
    private static final CountDownLatch ENTERED = new CountDownLatch(1);
    private static final CountDownLatch RELEASE = new CountDownLatch(1);
    private static final CountDownLatch REQUESTED = new CountDownLatch(1);
    private static int holdSeconds;

    private InitializationRace() { }

    static final class Lazy {
        static final int VALUE = initialize();

        private static int initialize() {
            LOG.log(System.Logger.Level.INFO, "初始化所有者进入，等待观察窗口结束");
            ENTERED.countDown();
            try {
                if (!RELEASE.await(holdSeconds + 5L, TimeUnit.SECONDS)) {
                    throw new IllegalStateException("观察窗口未按时释放");
                }
            } catch (InterruptedException exception) {
                Thread.currentThread().interrupt();
                throw new IllegalStateException("初始化线程被中断", exception);
            }
            return 73;
        }
    }

    private static void awaitPhase(CountDownLatch phase) throws InterruptedException {
        if (!phase.await(3, TimeUnit.SECONDS)) {
            throw new IllegalStateException("教学进程未及时到达预定阶段");
        }
    }

    public static void main(String[] args) throws InterruptedException {
        holdSeconds = args.length == 0 ? 0 : Integer.parseInt(args[0]);
        if (holdSeconds < 0 || holdSeconds > 30) {
            throw new IllegalArgumentException("观察窗口须为 0—30 秒");
        }
        Thread owner = Thread.ofPlatform().name("init-owner").unstarted(
                () -> System.out.println("owner.value=" + Lazy.VALUE));
        Thread contender = Thread.ofPlatform().name("init-contender").unstarted(() -> {
            REQUESTED.countDown();
            System.out.println("contender.value=" + Lazy.VALUE);
        });
        LOG.log(System.Logger.Level.INFO, "开始有界并发初始化观察");
        try {
            owner.start();
            awaitPhase(ENTERED);
            contender.start();
            awaitPhase(REQUESTED);
            // 此标记仅说明即将执行读取；实际等待状态需外部线程快照佐证。
            System.out.println("observe.pid=" + ProcessHandle.current().pid() + " seconds=" + holdSeconds);
            RELEASE.await(holdSeconds, TimeUnit.SECONDS);
        } finally {
            RELEASE.countDown();
            owner.join(5_000);
            contender.join(5_000);
            if (owner.isAlive() || contender.isAlive()) {
                owner.interrupt();
                contender.interrupt();
                LOG.log(System.Logger.Level.ERROR, "工作线程未按时收尾，已请求中断");
            }
        }
        LOG.log(System.Logger.Level.INFO, "有界并发初始化观察结束");
    }
}
